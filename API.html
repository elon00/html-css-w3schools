HTML APIs Reference for VS Code Development

DOM Manipulation APIs

Document Object Model (DOM)

// Basic DOM Selection document.getElementById('elementId') document.getElementsByClassName('className') document.getElementsByTagName('tagName') document.querySelector('selector') document.querySelectorAll('selector') // Element Creation and Manipulation document.createElement('tagName') element.appendChild(childElement) element.removeChild(childElement) element.innerHTML = 'content' element.textContent = 'text' element.setAttribute('attribute', 'value') element.getAttribute('attribute') element.classList.add('className') element.classList.remove('className') element.classList.toggle('className')
Example:
const button = document.createElement('button'); button.textContent = 'Click me'; button.addEventListener('click', function() { alert('Button clicked!'); }); document.body.appendChild(button);

Event Handling APIs

Event Listeners

// Adding Event Listeners element.addEventListener('event', function) element.addEventListener('event', function, options) // Common Events 'click', 'mouseover', 'mouseout', 'keydown', 'keyup' 'load', 'resize', 'scroll', 'submit', 'change' // Event Object Properties event.target event.currentTarget event.preventDefault() event.stopPropagation() event.type event.clientX, event.clientY
Example:
document.addEventListener('DOMContentLoaded', function() { const form = document.querySelector('#myForm'); form.addEventListener('submit', function(e) { e.preventDefault(); console.log('Form submitted'); }); });

Web Storage APIs

LocalStorage

// Store data localStorage.setItem('key', 'value') localStorage.setItem('key', JSON.stringify(object)) // Retrieve data localStorage.getItem('key') JSON.parse(localStorage.getItem('key')) // Remove data localStorage.removeItem('key') localStorage.clear() // Check storage localStorage.length localStorage.key(index)

SessionStorage

// Same methods as localStorage but data persists only for session sessionStorage.setItem('key', 'value') sessionStorage.getItem('key') sessionStorage.removeItem('key') sessionStorage.clear()

Fetch API

Making HTTP Requests

// Basic GET request fetch('https://api.example.com/data') .then(response => response.json()) .then(data => console.log(data)) .catch(error => console.error('Error:', error)); // POST request with data fetch('https://api.example.com/data', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ name: 'John', email: 'john@example.com' }) }) .then(response => response.json()) .then(data => console.log(data)); // Async/Await syntax async function fetchData() { try { const response = await fetch('https://api.example.com/data'); const data = await response.json(); return data; } catch (error) { console.error('Error:', error); } }

Canvas API

2D Drawing Context

// Get canvas context const canvas = document.getElementById('myCanvas'); const ctx = canvas.getContext('2d'); // Drawing methods ctx.fillRect(x, y, width, height) ctx.strokeRect(x, y, width, height) ctx.clearRect(x, y, width, height) // Path methods ctx.beginPath() ctx.moveTo(x, y) ctx.lineTo(x, y) ctx.arc(x, y, radius, startAngle, endAngle) ctx.stroke() ctx.fill() // Styling ctx.fillStyle = 'color' ctx.strokeStyle = 'color' ctx.lineWidth = number ctx.font = 'size family' ctx.fillText('text', x, y)

Geolocation API

// Check if geolocation is supported if (navigator.geolocation) { // Get current position navigator.geolocation.getCurrentPosition( function(position) { const lat = position.coords.latitude; const lon = position.coords.longitude; console.log(`Latitude: ${lat}, Longitude: ${lon}`); }, function(error) { console.error('Error getting location:', error); } ); // Watch position changes const watchId = navigator.geolocation.watchPosition( function(position) { // Handle position update } ); // Clear watch navigator.geolocation.clearWatch(watchId); }

File API

// File input handling const fileInput = document.getElementById('fileInput'); fileInput.addEventListener('change', function(e) { const files = e.target.files; for (let i = 0; i < files.length; i++) { const file = files[i]; // File properties console.log('Name:', file.name); console.log('Size:', file.size); console.log('Type:', file.type); console.log('Last Modified:', file.lastModified); // Read file content const reader = new FileReader(); reader.onload = function(e) { console.log('File content:', e.target.result); }; reader.readAsText(file); // reader.readAsDataURL(file); // For images // reader.readAsArrayBuffer(file); // For binary data } }); // Drag and Drop const dropZone = document.getElementById('dropZone'); dropZone.addEventListener('dragover', function(e) { e.preventDefault(); }); dropZone.addEventListener('drop', function(e) { e.preventDefault(); const files = e.dataTransfer.files; // Process files });

Web Workers API

// Main thread const worker = new Worker('worker.js'); // Send message to worker worker.postMessage({ command: 'process', data: [1, 2, 3, 4, 5] }); // Receive message from worker worker.addEventListener('message', function(e) { console.log('Result from worker:', e.data); }); // Handle errors worker.addEventListener('error', function(e) { console.error('Worker error:', e); }); // Terminate worker worker.terminate();
worker.js file:
// Inside worker.js self.addEventListener('message', function(e) { const { command, data } = e.data; if (command === 'process') { // Perform heavy computation const result = data.map(x => x * x); // Send result back to main thread self.postMessage(result); } });

Intersection Observer API

// Create observer const observer = new IntersectionObserver(function(entries) { entries.forEach(entry => { if (entry.isIntersecting) { console.log('Element is visible'); // Lazy load image, trigger animation, etc. } }); }, { threshold: 0.1, // Trigger when 10% visible rootMargin: '50px' // Trigger 50px before entering viewport }); // Start observing elements const targets = document.querySelectorAll('.lazy-load'); targets.forEach(target => observer.observe(target)); // Stop observing observer.unobserve(element); observer.disconnect();

Notification API

// Request permission Notification.requestPermission().then(function(permission) { if (permission === 'granted') { // Create notification const notification = new Notification('Hello!', { body: 'This is a notification', icon: '/icon.png', tag: 'unique-tag' }); // Handle events notification.onclick = function() { console.log('Notification clicked'); }; // Auto close after 5 seconds setTimeout(() => notification.close(), 5000); } }); // Check current permission console.log(Notification.permission); // 'granted', 'denied', or 'default'

WebSocket API

// Create WebSocket connection const socket = new WebSocket('ws://localhost:8080'); // Connection opened socket.addEventListener('open', function(event) { console.log('Connected to WebSocket server'); socket.send('Hello Server!'); }); // Listen for messages socket.addEventListener('message', function(event) { console.log('Message from server:', event.data); }); // Connection closed socket.addEventListener('close', function(event) { console.log('WebSocket connection closed'); }); // Handle errors socket.addEventListener('error', function(error) { console.error('WebSocket error:', error); }); // Send data socket.send(JSON.stringify({ type: 'message', content: 'Hello!' })); // Close connection socket.close();

Clipboard API

// Modern Clipboard API (requires HTTPS) // Write text to clipboard navigator.clipboard.writeText('Text to copy').then(function() { console.log('Text copied to clipboard'); }).catch(function(err) { console.error('Could not copy text:', err); }); // Read text from clipboard navigator.clipboard.readText().then(function(text) { console.log('Clipboard content:', text); }).catch(function(err) { console.error('Could not read clipboard:', err); }); // Legacy method (fallback) function copyToClipboard(text) { const textArea = document.createElement('textarea'); textArea.value = text; document.body.appendChild(textArea); textArea.select(); document.execCommand('copy'); document.body.removeChild(textArea); }

History API

// Navigate without page reload history.pushState({page: 'home'}, 'Home', '/home'); history.replaceState({page: 'about'}, 'About', '/about'); // Go back/forward history.back(); history.forward(); history.go(-2); // Go back 2 pages // Listen for navigation window.addEventListener('popstate', function(event) { console.log('Navigation occurred:', event.state); }); // Get current state console.log(history.state);

Page Visibility API

// Check if page is visible if (document.visibilityState === 'visible') { console.log('Page is visible'); } else { console.log('Page is hidden'); } // Listen for visibility changes document.addEventListener('visibilitychange', function() { if (document.hidden) { // Page is hidden - pause video, stop timers, etc. console.log('Page is now hidden'); } else { // Page is visible - resume activities console.log('Page is now visible'); } });

Common HTML5 Form APIs

// Form validation const form = document.querySelector('form'); const input = document.querySelector('input'); // Check validity console.log(input.validity.valid); console.log(input.validity.valueMissing); console.log(input.validity.typeMismatch); // Custom validation input.setCustomValidity('Custom error message'); input.setCustomValidity(''); // Clear custom validity // Form data handling const formData = new FormData(form); formData.append('key', 'value'); formData.get('key'); formData.has('key'); formData.delete('key'); // Convert to URL parameters const params = new URLSearchParams(formData); console.log(params.toString());
